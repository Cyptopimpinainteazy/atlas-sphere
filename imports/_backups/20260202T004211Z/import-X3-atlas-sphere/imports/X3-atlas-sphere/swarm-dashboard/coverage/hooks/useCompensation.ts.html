      baseCompensation = model.rate_per_asset * metrics.total_assets;

function getTierMultiplier(tier: string): number {
  switch (tier) {
    case 'Bronze': return 1.0;
    case 'Silver': return 1.1;
    case 'Gold': return 1.25;
    case 'Platinum': return 1.5;
    default: return 1.0;
  }
}
 
function calculateQualityBonus(metrics: ContributorMetrics): number {
